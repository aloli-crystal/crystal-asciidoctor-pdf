=== Crystal

[source,crystal]
----
require "kemal"

class UserHandler < HTTP::Handler
  def call(context)
    if context.request.path == "/api/users"
      users = User.all
      context.response.content_type = "application/json"
      context.response.print users.to_json
    else
      call_next(context)
    end
  end
end

# Démarrage du serveur
Kemal.config.port = 3000
Kemal.run
----

=== SQL

[source,sql]
----
-- Création de la table des utilisateurs
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(50) DEFAULT 'lecteur',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Requête avec jointures
SELECT u.email, p.name AS project_name, COUNT(t.id) AS task_count
FROM users u
JOIN team_members tm ON u.id = tm.user_id
JOIN projects p ON tm.project_id = p.id
LEFT JOIN tasks t ON p.id = t.project_id
GROUP BY u.email, p.name
ORDER BY task_count DESC
LIMIT 10;
----
